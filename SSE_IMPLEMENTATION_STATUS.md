```typescript
async createPurchaseOrder(orderData) {
  let createdOrder;
  await prisma.$transaction(async (tx) => {

      data: {
        orderNumber,
        productId: orderData.productId,
        supplierId: orderData.supplierId,

      id: createdOrder.id,

      productId: createdOrder.productId,
      warehouseId: createdOrder.warehouseId,
```

```typescript
// Order arrival processing
async processOrderArrival(purchaseOrderId: string) {
  await prisma.$transaction(async (tx) => {
    // Update order status to ARRIVED
    await tx.purchaseOrder.update({
      where: { id: purchaseOrderId },
      data: { status: 'ARRIVED', actualArrivalDate: new Date() }
    });

    // Update product stock
    await tx.productStock.upsert({ ... });

    // Record stock movement
    await tx.stockMovement.create({ ... });
  });

  const order = await prisma.purchaseOrder.findUnique({
    where: { id: purchaseOrderId },
    include: { product: true, warehouse: true }
  });

  // Update warehouse stock count
  await capacityManager.updateWarehouseStock(order.warehouseId);

  // Emit order:updated event
  appEvents.emitOrderUpdated({
    id: order.id,
    orderNumber: order.orderNumber,
    status: order.status,
  });

  // Emit stock:update event
  const updatedStock = await prisma.productStock.findUnique({ ... });
  appEvents.emitStockUpdate({
    productId: order.productId,
    warehouseId: order.warehouseId,
    quantity: updatedStock.quantity,
    productName: order.product?.name,
    warehouseName: order.warehouse?.name,
  });
}
```
**Trigger:** `POST /api/purchase-orders/:id/arrive`
**Database:** ✅ All updates saved before emissions
**Tested:** ✅ Yes (triggers 3 events!)

### 6. warehouse:updated Event (1 emission point)

#### Location: capacityManager.service.ts:170
```typescript
async updateWarehouseStock(warehouseId: string) {
  const totalStock = await prisma.productStock.aggregate({
    where: { warehouseId },
    _sum: { quantity: true }
  });

  const warehouse = await prisma.warehouse.update({
    where: { id: warehouseId },
    data: { currentStock: totalStock._sum.quantity || 0 }
  });

  const percentage = warehouse.capacity > 0
    ? (warehouse.currentStock / warehouse.capacity) * 100
    : 0;

  appEvents.emitWarehouseUpdated({
    id: warehouse.id,
    name: warehouse.name,
    currentStock: warehouse.currentStock,
    capacity: warehouse.capacity,
    percentage,
  });
}
```
**Trigger:** Called after order arrival (part of processOrderArrival)
**Database:** ✅ Warehouse updated before emission
**Tested:** ✅ Yes (with order arrival)
